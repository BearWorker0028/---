## 三、 後端 API 與 Modbus 採集解析範例 (Python)

```python
import struct
from pymodbus.client import ModbusSerialClient

def read_spm3_meter(client, slave_id):
    # 讀取 1032 ~ 1084 (線電壓、電流、功率、PF)
    resp = client.read_input_registers(address=1032, count=52, device_id=slave_id)
    # 讀取 1182 ~ 1183 (總電量 kWh)
    resp_kwh = client.read_input_registers(address=1182, count=2, device_id=slave_id)
    
    if resp.isError() or resp_kwh.isError():
        return None

    r = resp.registers
    r_kwh = resp_kwh.registers

    def parse_float(r0, r1):
        # Low word first (r1, r0)
        return round(struct.unpack('>f', struct.pack('>HH', r1, r0))[0], 2)

    return {
        'voltage_rs': parse_float(r[0], r[1]),     # 1032
        'voltage_st': parse_float(r[2], r[3]),     # 1034
        'voltage_tr': parse_float(r[4], r[5]),     # 1036
        'voltage_avg': parse_float(r[6], r[7]),    # 1038
        'current_r':  parse_float(r[20], r[21]),   # 1052
        'current_s':  parse_float(r[22], r[23]),   # 1054
        'current_t':  parse_float(r[24], r[25]),   # 1056
        'current_avg': parse_float(r[28], r[29]),  # 1060
        'power_total': parse_float(r[42], r[43]),  # 1074 (kW)
        'power_factor': parse_float(r[50], r[51]), # 1082 (PF)
        'energy_total': parse_float(r_kwh[0], r_kwh[1]) # 1182 (kWh)
    }
```
